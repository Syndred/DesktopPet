/** Real production app replay. The supplied isolated profile must already contain native-dialog imports.
 * Usage from repo root: node apps/pc-app/runtime/features/resources/verification/smoke.electron.mjs <profile> [output]
 * This does not substitute a fake dialog or overwrite source/user assets.
 */
import { _electron as electron } from 'playwright';
import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { mkdirSync, readFileSync, writeFileSync, existsSync } from 'node:fs';
import { resolve, join } from 'node:path';

assert.ok(process.argv[2], 'Pass an isolated acceptance profile with real native-dialog imports.');
const profile = resolve(process.argv[2]);
assert.ok(existsSync(join(profile, 'lingjing.sqlite')), 'The acceptance profile must already exist.');
const root = resolve('.'), output = resolve(process.argv[3] || 'artifacts/resources/final');
mkdirSync(output, { recursive: true });
const env = { ...process.env, PET_USER_DATA_DIR: profile }; delete env.ELECTRON_RUN_AS_NODE;
const application = await electron.launch({ args: [root], env });
const errors = [], consoleErrors = [];
const hash = bytes => createHash('sha256').update(bytes).digest('hex');
try {
  const page = await application.firstWindow();
  page.on('pageerror', error => errors.push(error.stack || error.message));
  page.on('console', message => { if (message.type() === 'error') consoleErrors.push(message.text()); });
  await page.getByRole('complementary').getByRole('button', { name: '角色资源', exact: true }).click();
  await page.locator('.resource-card').first().waitFor();
  const response = await page.evaluate(() => window.lingjing.invoke('resources', 'list'));
  assert.equal(response.ok, true);
  const fish = response.data.find(resource => resource.origin === 'import' && resource.kind === 'spritesheet');
  const model = response.data.find(resource => resource.origin === 'import' && resource.kind === 'model3d');
  assert.ok(fish && model, 'Import a real 2D pack and an existing GLB through the native UI first.');
  assert.equal(response.data.filter(resource => resource.origin === 'legacy').length, 17);
  assert.ok(!JSON.stringify(response).includes(profile), 'No local paths may cross the bridge.');
  const stored = JSON.parse(readFileSync(resolve(profile, '..', 'sqlite-before-restart.json'), 'utf8'));
  for (const resource of [fish, model]) {
    const pref = stored.preferences.find(row => row.resource_id === resource.id);
    assert.ok(pref); assert.deepEqual(resource.calibration, JSON.parse(pref.calibration_json));
    assert.deepEqual(resource.actions, JSON.parse(pref.actions_json)); assert.equal(resource.revision, pref.revision);
  }
  const choose = resource => page.locator(`.resource-card[data-resource-id="${resource.id}"]`).click();
  await choose(fish); await page.locator('.resource-stage[data-status="ready"]').waitFor();
  await page.getByRole('button', { name: '双实例对照', exact: true }).click();
  await page.getByLabel('右侧独立动作', { exact: true }).selectOption('sleep');
  await page.getByRole('button', { name: '暂停动画', exact: true }).click();
  await page.locator('.resource-stage[data-status="ready"]').nth(1).waitFor();
  await page.locator('.resource-stage[data-status="ready"]').nth(0).waitFor();
  await page.locator('.resource-stage-pair').scrollIntoViewIfNeeded();
  await new Promise(done => setTimeout(done, 250));
  const before = await Promise.all([page.locator('.resource-stage').nth(0).screenshot(), page.locator('.resource-stage').nth(1).screenshot()]);
  await new Promise(done => setTimeout(done, 450));
  const after = await Promise.all([page.locator('.resource-stage').nth(0).screenshot(), page.locator('.resource-stage').nth(1).screenshot()]);
  const independent = { primaryPaused: hash(before[0]) === hash(after[0]), secondaryAnimated: hash(before[1]) !== hash(after[1]) };
  assert.deepEqual(independent, { primaryPaused: true, secondaryAnimated: true });
  const instances = await page.locator('.resource-stage').evaluateAll(nodes => nodes.map(node => ({ petId: node.dataset.petId, resourceId: node.dataset.resourceId, clip: node.dataset.clip })));
  assert.equal(new Set(instances.map(item => item.resourceId)).size, 1); assert.equal(new Set(instances.map(item => item.petId)).size, 2);
  await page.screenshot({ path: join(output, 'independent-2d.png') });
  await page.getByLabel('预览动作', { exact: true }).selectOption('skill');
  await page.locator('.resource-stage[data-pet-id="resource-preview-a"][data-fallback="true"][data-clip="idle"]').waitFor();
  await page.getByRole('button', { name: '收起双实例对照', exact: true }).click();
  await choose(model); await page.locator('.resource-stage[data-status="ready"]').waitFor();
  await page.getByRole('button', { name: '战斗', exact: true }).click();
  await page.locator('.resource-stage[data-status="ready"]').waitFor();
  await page.getByRole('button', { name: '桌面', exact: true }).click();
  await page.locator('.resource-stage[data-status="ready"]').waitFor();
  const loadedModel = await page.locator('model-viewer').evaluate(element => ({
    loaded: element.loaded, source: element.src, animation: element.animationName, animations: element.availableAnimations,
    orientation: element.orientation, materialTexture: element.model.materials.some(material => Boolean(material.pbrMetallicRoughness.baseColorTexture.texture)),
  }));
  assert.equal(loadedModel.loaded, true); assert.equal(loadedModel.materialTexture, true);
  assert.ok(loadedModel.source.includes(model.id));
  assert.equal(loadedModel.animation, model.actions.idle);
  await page.locator('.resources-heading').scrollIntoViewIfNeeded();
  await page.screenshot({ path: join(output, 'textured-3d.png') });
  assert.deepEqual(errors, []); assert.deepEqual(consoleErrors, []);
  const result = { at: new Date().toISOString(), nativeImportsPreviouslyPerformed: true, restart: true,
    resources: { count: response.data.length, legacy: 17, fish, model }, independent, instances, loadedModel,
    errors, consoleErrors, bridgeIsolated: await page.evaluate(() => typeof window.require === 'undefined'),
    originalImportReceipts: stored.imports, screenshots: ['independent-2d.png', 'textured-3d.png'] };
  writeFileSync(join(output, 'runtime-evidence.json'), JSON.stringify(result, null, 2));
  console.log(JSON.stringify({ ok: true, output, resourceIds: [fish.id, model.id], independent, modelLoaded: loadedModel.loaded, textures: loadedModel.materialTexture, errors }));
} finally { await application.close(); }
